```rust
pub fn border_ring(
    rect: &Rect,
    radii: &[(f32, f32); 4],
    width: f32,
    color: [f32; 4],
) -> (Vec<[f32; 2]>, Vec<[f32; 2]>, Vec<[f32; 4]>, Vec<u32>) {
    if width <= 0.0 || rect.w <= 0.0 || rect.h <= 0.0 {
        return (Vec::new(), Vec::new(), Vec::new(), Vec::new());
    }
    let w = width.min(rect.w * 0.5).min(rect.h * 0.5); // 防内轮廓交叉
    let (x, y, rw, rh) = (rect.x, rect.y, rect.w, rect.h);
    let has_radius = radii.iter().any(|(rx, ry)| *rx > 0.0 || *ry > 0.0);
    // ponytail: 圆角边框分段留 Task 5 圆角 SDF 一起补；此处有 radius 时按直角退化。
    // 升级路径：外/内轮廓各走 rounded_rect 圆弧顶点，环形带连两轮廓。
    let _ = has_radius;

    // 外轮廓 4 角（TL, TR, BR, BL），内轮廓 4 角（同序缩进 w）。
    let outer = [
        [x, y],
        [x + rw, y],
        [x + rw, y + rh],
        [x, y + rh],
    ];
    let inner = [
        [x + w, y + w],
        [x + rw - w, y + w],
        [x + rw - w, y + rh - w],
        [x + w, y + rh - w],
    ];
    let mut verts = Vec::with_capacity(8);
    verts.extend_from_slice(&outer);
    verts.extend_from_slice(&inner);
    let uvs = vec![[0.0, 0.0]; 8]; // 纯色，不采样
    let colors = vec![color; 8];
    // 每边 2 三角连 outer[i]/outer[i+1]/inner[i+1]/inner[i]（环形带）。
    let mut indices = Vec::with_capacity(24);
    for i in 0..4 {
        let ni = (i + 1) % 4;
        let (oi, oni) = (i as u32, ni as u32);
        let (ii, ini) = ((i + 4) as u32, (ni + 4) as u32);
        indices.extend_from_slice(&[oi, oni, ini, oi, ini, ii]);
    }
    (verts, uvs, colors, indices)
}
```

```rust
/// 4 角独立色的 quad（2 色线性渐变用）。colors 顺序与 quad 顶点一致：TL, TR, BR, BL。
pub fn quad_gradient(
    rect: &Rect,
    colors: [[f32; 4]; 4],
    uv_min: [f32; 2],
    uv_max: [f32; 2],
) -> (Vec<[f32; 2]>, Vec<[f32; 2]>, Vec<[f32; 4]>, Vec<u32>) {
    let verts = vec![
        [rect.x, rect.y],
        [rect.x + rect.w, rect.y],
        [rect.x + rect.w, rect.y + rect.h],
        [rect.x, rect.y + rect.h],
    ];
    let (umin, vmin) = (uv_min[0], uv_min[1]);
    let (umax, vmax) = (uv_max[0], uv_max[1]);
    let uvs = vec![[umin, vmin], [umax, vmin], [umax, vmax], [umin, vmax]];
    let colors = colors.to_vec();
    let indices = vec![0, 1, 2, 0, 2, 3];
    (verts, uvs, colors, indices)
}
```

新增辅助函数 `parse_linear_gradient_2(style: &mut ResolvedStyle, inner: &str) -> bool`：

```rust
/// 解析 linear-gradient 内部 "to right, #f00, #00f"。
/// 2 色 + 4 方向（to right/left/top/bottom）；多 stop（>2 色）或斜角度 → false（编译期报错由打包器）。
fn parse_linear_gradient_2(style: &mut ResolvedStyle, inner: &str) -> bool {
    let parts: Vec<&str> = inner.split(',').map(str::trim).collect();
    // 第一段是方向 "to <dir>"，其余是色 stop。
    if parts.len() < 3 {
        return false; // 缺方向或不足 2 色
    }
    let dir = match parts[0] {
        "to right" => GradientDir::ToRight,
        "to left" => GradientDir::ToLeft,
        "to top" => GradientDir::ToTop,
        "to bottom" => GradientDir::ToBottom,
        _ => return false, // 斜角度（45deg 等）或未知 → 围栏外，apply_decl 返 false
    };
    if parts.len() > 3 {
        return false; // 多 stop（>2 色）→ 围栏外拒收（不让 AI 误以为支持）
    }
    let color_a = match parse_color(parts[1]) {
        Some(c) => c,
        None => return false,
    };
    let color_b = match parse_color(parts[2]) {
        Some(c) => c,
        None => return false,
    };
    style.background_gradient = Some(Gradient2 { color_a, color_b, dir });
    true
}
```

> 注意：多 stop/斜角度返 `false` = apply_decl 静默忽略。**但 spec §4.3 要求编译期报错**。静默 vs 报错的区别在打包器层：apply_decl 返 false 时打包器是否报错取决于打包器对"围栏外属性"的策略。现有围栏外属性（如 clip-path）是静默忽略（apply_decl 返 false 不报错）。
>
> **决策**：linear-gradient 多 stop/斜角度与现有"围栏外 CSS 静默忽略"一致（返 false），不特殊报错——因为 `background: linear-gradient(45deg,...)` 整体是合法 CSS，只是 LoomGUI 不支持该形态，静默忽略 background（不渲染渐变）比报错更符合现有围栏语义。**spec §4.3 的"编译期报错"在此调整为"静默忽略（不渲染渐变）"**，与 clip-path 等一致。文档（fence.md）注明 linear-gradient 仅支持 2 色 4 方向。

```rust
/// box-shadow 几何近似：比 rect 外扩 spread 的圆角 quad（直角退化），四周顶点 alpha 渐隐。
/// 无 blur（真实 blur 需离屏 RT，排 v1.14+）。独立 RenderNode 画在节点下层。
pub fn box_shadow_quad(
    rect: &Rect,
    radii: &[(f32, f32); 4],
    spread: f32,
    color: [f32; 4],
) -> (Vec<[f32; 2]>, Vec<[f32; 2]>, Vec<[f32; 4]>, Vec<u32>) {
    let _ = radii; // ponytail: 圆角阴影随 Task 5 圆角 SDF 补，先直角
    let outer = Rect {
        x: rect.x - spread,
        y: rect.y - spread,
        w: rect.w + 2.0 * spread,
        h: rect.h + 2.0 * spread,
    };
    if outer.w <= 0.0 || outer.h <= 0.0 {
        return (Vec::new(), Vec::new(), Vec::new(), Vec::new());
    }
    crate::render::mesh::quad(&outer, color, [0.0, 0.0], [0.0, 0.0])
}
```

```rust
/// 九宫格 slice % resolve：值 ∈ (0,1) 视为比例，乘 src 尺寸成像素；≥1 视为像素不动。
/// 修 v1.6 bug：parse 把 25% 存 0.25，render 期无 resolve 当像素用 → 0.25px 坍缩。
pub fn resolve_slice_percent(s: &SliceInsets, src_w: f32, src_h: f32) -> SliceInsets {
    let rx = |v: f32, src: f32| if v > 0.0 && v < 1.0 { v * src } else { v };
    SliceInsets {
        top: rx(s.top, src_h),
        bottom: rx(s.bottom, src_h),
        left: rx(s.left, src_w),
        right: rx(s.right, src_w),
    }
}
```

> 0 < v < 1 视为比例（25% = 0.25），≥1 视为像素。注意：合法像素切片最小 1，故 (0,1) 区间无歧义。

```rust
//! 对标 RmlUi ConvolutionFilter（取 max=dilation / 累加=sum=高斯），但用 etagere 增量
//! atlas 规避 RmlUi 全量重建纹理的坑。

/// 单个文字效果配置。
#[derive(Debug, Clone, Copy, PartialEq)]
pub enum FontEffect {
    /// 阴影：offset + 高斯 blur。text-shadow。
    Shadow { ox: f32, oy: f32, blur: f32, color: [f32; 4] },
    /// 描边：内侧吃字（erode）。-webkit-text-stroke。
    Stroke { w: f32, color: [f32; 4] },
    /// 发光：dilate + 高斯 blur。font-effect:glow。
    Glow { w: f32, color: [f32; 4] },
    /// 模糊：可分离高斯两 pass。font-effect:blur。
    Blur { w: f32 },
}

/// effect_sig hash 输入 = (type, params)。per-size 已在 GlyphKey 不进 sig。
pub fn effect_sig(effects: &[FontEffect]) -> u64 {
    use std::collections::hash_map::DefaultHasher;
    use std::hash::{Hash, Hasher};
    let mut h = DefaultHasher::new();
    for e in effects {
        std::mem::discriminant(e).hash(&mut h);
        match e {
            FontEffect::Shadow { ox, oy, blur, color } => {
                ox.to_bits().hash(&mut h); oy.to_bits().hash(&mut h);
                blur.to_bits().hash(&mut h); color.hash(&mut h);
            }
            FontEffect::Stroke { w, color } => { w.to_bits().hash(&mut h); color.hash(&mut h); }
            FontEffect::Glow { w, color } => { w.to_bits().hash(&mut h); color.hash(&mut h); }
            FontEffect::Blur { w } => { w.to_bits().hash(&mut h); }
        }
    }
    h.finish()
}
```

```rust
/// 圆形 dilation：每个输出像素 = 邻域（radius 圆内）alpha 的 max。向外扩边界 ceil(radius)。
/// glow 用（膨胀亮边）。对标 RmlUi FilterOperation::Dilation。
pub fn dilate(p: &[u8], w: u32, h: u32, radius: f32) -> (Vec<u8>, u32, u32) {
    convolve(p, w, h, radius, true)
}

/// 圆形 erosion：每个输出像素 = 邻域 alpha 的 min。向内缩。
/// stroke 内侧吃字用（dilation 反向）。
pub fn erode(p: &[u8], w: u32, h: u32, radius: f32) -> (Vec<u8>, u32, u32) {
    convolve(p, w, h, radius, false)
}

/// convolve 通用：dilation=true 取 max，false 取 min。圆形核，radius 内 weight=1。
fn convolve(p: &[u8], w: u32, h: u32, radius: f32, dilation: bool) -> (Vec<u8>, u32, u32) {
    let r = radius.ceil() as i32;
    if r <= 0 {
        return (p.to_vec(), w, h);
    }
    let nw = (w as i32 + 2 * r) as u32;
    let nh = (h as i32 + 2 * r) as u32;
    let mut out = vec![0u8; (nw * nh) as usize];
    let rf = radius * radius;
    for oy in 0..nh as i32 {
        for ox in 0..nw as i32 {
            let mut best: u8 = if dilation { 0 } else { 255 };
            for dy in -r..=r {
                for dx in -r..=r {
                    if (dx * dx + dy * dy) as f32 > rf { continue; } // 圆形核
                    let sx = ox - r + dx;
                    let sy = oy - r + dy;
                    if sx < 0 || sy < 0 || sx >= w as i32 || sy >= h as i32 { continue; }
                    let v = p[(sy as u32 * w + sx as u32) as usize];
                    best = if dilation { best.max(v) } else { best.min(v) };
                }
            }
            out[(oy as u32 * nw + ox as u32) as usize] = best;
        }
    }
    (out, nw, nh)
}

/// 可分离高斯 blur：水平 + 垂直两 pass。同尺寸（不扩边界）。
/// 标准权重 exp(-x²/(2σ²))（不用 RmlUi 非标准 sqrt(x²) 形态）。
pub fn gaussian_blur(p: &[u8], w: u32, h: u32, sigma: f32) -> Vec<u8> {
    if sigma <= 0.0 || w == 0 || h == 0 {
        return p.to_vec();
    }
    let r = (sigma * 3.0).ceil() as i32; // 3σ 截断
    // 1D 权重
    let mut weights = Vec::with_capacity((2 * r + 1) as usize);
    let two_sigma2 = 2.0 * sigma * sigma;
    let mut sum = 0.0;
    for i in -r..=r {
        let wv = (-((i * i) as f32) / two_sigma2).exp();
        weights.push(wv);
        sum += wv;
    }
    for wv in &mut weights { *wv /= sum; } // 归一化
    // 水平 pass
    let mut tmp = vec![0.0f32; (w * h) as usize];
    for y in 0..h as i32 {
        for x in 0..w as i32 {
            let mut acc = 0.0;
            for (k, &wv) in weights.iter().enumerate() {
                let sx = (x + k as i32 - r).clamp(0, w as i32 - 1);
                acc += p[(y as u32 * w + sx as u32) as usize] as f32 * wv;
            }
            tmp[(y as u32 * w + x as u32) as usize] = acc;
        }
    }
    // 垂直 pass
    let mut out = vec![0u8; (w * h) as usize];
    for y in 0..h as i32 {
        for x in 0..w as i32 {
            let mut acc = 0.0;
            for (k, &wv) in weights.iter().enumerate() {
                let sy = (y + k as i32 - r).clamp(0, h as i32 - 1);
                acc += tmp[(sy as u32 * w + x as u32) as usize] * wv;
            }
            out[(y as u32 * w + x as u32) as usize] = acc.clamp(0.0, 255.0) as u8;
        }
    }
    out
}
```

- [ ] **Step 6: 实现 apply_effect（组合算法）**

```rust
/// 对单字形 R8 位图应用一个 effect，返回后处理位图（可能扩边界）。
/// 颜色（color）不在此处理——位图只存 alpha，颜色由 build 期 vertex color 提供
/// （同 v1.6 premultiplied alpha 约定）。此处 color 仅参与 sig 区分。
pub fn apply_effect(p: &[u8], w: u32, h: u32, effect: &FontEffect) -> (Vec<u8>, u32, u32) {
    match effect {
        FontEffect::Shadow { blur, .. } => {
            // shadow = clone + blur + offset。offset 在 build 期偏移 quad，位图只 blur。
            if *blur > 0.0 {
                let sigma = blur / 2.0;
                (gaussian_blur(p, w, h, sigma), w, h)
            } else {
                (p.to_vec(), w, h)
            }
        }
        FontEffect::Stroke { w: sw, .. } => {
            // 内侧吃字 = erode（向内缩 sw 像素）。erode 扩边界但语义是内缩——
            // stroke 描边色填充 erode 出的边界环：返 erode 结果，build 期描边层用之。
            erode(p, w, h, *sw)
        }
        FontEffect::Glow { w: gw, .. } => {
            // glow = dilate + blur。先 dilate 膨胀亮边，再 blur 晕开。
            let (d, dw, dh) = dilate(p, w, h, *gw);
            let sigma = *gw / 2.0;
            (gaussian_blur(&d, dw, dh, sigma), dw, dh)
        }
        FontEffect::Blur { w: bw } => {
            let sigma = *bw / 2.0;
            (gaussian_blur(p, w, h, sigma), w, h)
        }
    }
}
```

```rust
"text-decoration" => {
    // CSS3 shorthand: "underline dashed red 2px"（line style color thickness 任意序）
    let mut deco = RichDeco::default();
    for tok in val.split_whitespace() {
        match tok {
            "underline" => deco.lines.0 |= TextDecoLines::UNDERLINE.0,
            "line-through" => deco.lines.0 |= TextDecoLines::LINE_THROUGH.0,
            "overline" => deco.lines.0 |= TextDecoLines::OVERLINE.0,
            "dashed" => deco.style = TextDecoStyle::Dashed,
            "dotted" => deco.style = TextDecoStyle::Dotted,
            "double" => deco.style = TextDecoStyle::Double,
            "solid" => deco.style = TextDecoStyle::Solid,
            "none" => deco.lines = TextDecoLines::NONE,
            _ => {
                if let Some(c) = parse_color(tok) { deco.color = Some(c); }
                else if let Ok(t) = tok.parse::<f32>() { deco.thickness = Some(t); }
            }
        }
    }
    eff.deco = deco;
}
```
